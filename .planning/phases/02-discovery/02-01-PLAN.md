**src/lib/repositories/search.repository.ts** — raw SQL queries against the innovation_records table.

The `search_vector` column is already maintained by `trg_ir_search_vector` (plan 01-01). Use `plainto_tsquery` for natural-language queries and `ts_rank` for ranking, exactly as specified in TechArch §2.4:

```typescript
import { sql } from 'kysely';
import { db } from '@/lib/db/client';
import type { CatalogCardData } from '@/lib/db/types';

export interface RawSearchParams {
  query?: string;           // user-supplied text (may be empty/null → browse mode)
  maturity?: string[];
  mission_areas?: string[];
  technology_areas?: string[];
  review_statuses?: string[];
  contributing_offices?: string[];
  reuse_potential?: string;
  has_artifacts?: boolean;
  publication_states?: string[];  // public callers always get ['published']
  page: number;
  page_size: number;
  sort: 'relevance' | 'last_reviewed_desc' | 'title_asc';
}

export interface SearchRow extends CatalogCardData {
  rank: number;
  total_count: number;
}

export async function executeSearch(params: RawSearchParams): Promise<SearchRow[]> {
  // Build a Kysely query using raw SQL for the tsvector parts
  let query = db
    .selectFrom('innovation_records as ir')
    .select([
      'ir.id', 'ir.slug', 'ir.title', 'ir.summary',
      'ir.maturity', 'ir.review_statuses', 'ir.contributing_offices',
      'ir.last_reviewed_date', 'ir.engagement_indicator', 'ir.publication_state',
      'ir.technology_areas', 'ir.mission_areas', 'ir.reuse_potential',
      sql<number>`COUNT(*) OVER()`.as('total_count'),
    ]);

  // Text search — use plainto_tsquery for natural language (per TechArch §2.4)
  if (params.query && params.query.trim().length >= 2) {
    query = query
      .select(sql<number>`ts_rank(ir.search_vector, plainto_tsquery('english', ${params.query}))`.as('rank'))
      .where(sql`ir.search_vector @@ plainto_tsquery('english', ${params.query})`);
  } else {
    // Browse mode (no query): all published records, rank = 0
    query = query.select(sql<number>`0`.as('rank'));
  }

  // Always filter to published (public endpoint)
  const states = params.publication_states ?? ['published'];
  query = query.where('ir.publication_state', 'in', states);

  // Optional filters — GIN array operators for array fields
  if (params.maturity && params.maturity.length > 0) {
    query = query.where('ir.maturity', 'in', params.maturity);
  }
  if (params.mission_areas && params.mission_areas.length > 0) {
    query = query.where(sql`ir.mission_areas && ${sql.val(params.mission_areas)}::text[]`);
  }
  if (params.technology_areas && params.technology_areas.length > 0) {
    query = query.where(sql`ir.technology_areas && ${sql.val(params.technology_areas)}::text[]`);
  }
  if (params.review_statuses && params.review_statuses.length > 0) {
    query = query.where(sql`ir.review_statuses && ${sql.val(params.review_statuses)}::text[]`);
  }
  if (params.contributing_offices && params.contributing_offices.length > 0) {
    query = query.where(sql`ir.contributing_offices && ${sql.val(params.contributing_offices)}::text[]`);
  }
  if (params.reuse_potential) {
    query = query.where('ir.reuse_potential', '=', params.reuse_potential);
  }
  if (params.has_artifacts === true) {
    query = query.where(eb =>
      eb.exists(
        eb.selectFrom('artifacts').select('artifact_id').whereRef('artifacts.record_id', '=', 'ir.id')
      )
    );
  }

  // Sorting
  if (params.sort === 'relevance' && params.query) {
    query = query.orderBy(sql`rank`, 'desc').orderBy('ir.last_reviewed_date', 'desc');
  } else if (params.sort === 'title_asc') {
    query = query.orderBy('ir.title', 'asc');
  } else {
    // last_reviewed_desc (default for browse mode)
    query = query.orderBy('ir.last_reviewed_date', 'desc');
  }

  // Pagination
  const offset = (params.page - 1) * params.page_size;
  query = query.limit(params.page_size).offset(offset);

  return query.execute() as unknown as SearchRow[];
}

export interface FacetEntry { value: string; count: number; }
export interface FacetCounts {
  maturity: FacetEntry[];
  mission_areas: FacetEntry[];
  technology_areas: FacetEntry[];
  review_statuses: FacetEntry[];
  contributing_offices: FacetEntry[];
  reuse_potential: FacetEntry[];
}

export async function executeFacetCounts(): Promise<FacetCounts> {
  // Unnest array fields to count distinct values among published records
  const [maturity, mission_areas, technology_areas, review_statuses, contributing_offices, reuse_potential] = await Promise.all([
    db.selectFrom('innovation_records')
      .select(['maturity', sql<number>`COUNT(*)`.as('count')])
      .where('publication_state', '=', 'published')
      .where('maturity', 'is not', null)
      .groupBy('maturity')
      .orderBy('count', 'desc')
      .execute(),

    db.selectFrom(
      sql`(SELECT unnest(mission_areas) AS value FROM innovation_records WHERE publication_state = 'published') t`.as('t')
    )
      .select(['value', sql<number>`COUNT(*)`.as('count')])
      .groupBy('value')
      .orderBy('count', 'desc')
      .execute(),

    db.selectFrom(
      sql`(SELECT unnest(technology_areas) AS value FROM innovation_records WHERE publication_state = 'published') t`.as('t')
    )
      .select(['value', sql<number>`COUNT(*)`.as('count')])
      .groupBy('value')
      .orderBy('count', 'desc')
      .execute(),

    db.selectFrom(
      sql`(SELECT unnest(review_statuses) AS value FROM innovation_records WHERE publication_state = 'published') t`.as('t')
    )
      .select(['value', sql<number>`COUNT(*)`.as('count')])
      .groupBy('value')
      .orderBy('count', 'desc')
      .execute(),

    db.selectFrom(
      sql`(SELECT unnest(contributing_offices) AS value FROM innovation_records WHERE publication_state = 'published') t`.as('t')
    )
      .select(['value', sql<number>`COUNT(*)`.as('count')])
      .groupBy('value')
      .orderBy('count', 'desc')
      .execute(),

    db.selectFrom('innovation_records')
      .select(['reuse_potential', sql<number>`COUNT(*)`.as('count')])
      .where('publication_state', '=', 'published')
      .where('reuse_potential', 'is not', null)
      .groupBy('reuse_potential')
      .orderBy('count', 'desc')
      .execute(),
  ]);

  return {
    maturity: maturity.map(r => ({ value: r.maturity as string, count: Number(r.count) })),
    mission_areas: mission_areas.map(r => ({ value: r.value as string, count: Number(r.count) })),
    technology_areas: technology_areas.map(r => ({ value: r.value as string, count: Number(r.count) })),
    review_statuses: review_statuses.map(r => ({ value: r.value as string, count: Number(r.count) })),
    contributing_offices: contributing_offices.map(r => ({ value: r.value as string, count: Number(r.count) })),
    reuse_potential: reuse_potential.map(r => ({ value: r.reuse_potential as string, count: Number(r.count) })),
  };
}
```

```typescript
import { executeSearch, executeFacetCounts } from '@/lib/repositories/search.repository';
import type { SearchRow, FacetCounts, RawSearchParams } from '@/lib/repositories/search.repository';

// Canonical filter value sets for validation (F2.3)
const VALID_MATURITY_VALUES = ['idea', 'evaluated_idea', 'experiment_poc', 'prototype_pilot', 'production_validated', 'archived_retired'] as const;
const VALID_REUSE_POTENTIAL = ['high', 'moderate', 'low', 'not_assessed'] as const;
const VALID_SORTS = ['relevance', 'last_reviewed_desc', 'title_asc'] as const;

export interface SearchParams {
  q?: string;
  mission_areas?: string[];
  technology_areas?: string[];
  problem_type_tags?: string[];
  maturity?: string[];
  review_statuses?: string[];
  contributing_offices?: string[];
  reuse_potential?: string;
  has_artifacts?: boolean;
  publication_state?: string[];
  page?: number;
  page_size?: number;
  sort?: string;
}

export type { SearchRow, FacetCounts };

export interface SearchResult {
  records: Omit<SearchRow, 'rank' | 'total_count'>[];
  meta: {
    page: number;
    page_size: number;
    total: number;
    query: string;
    active_filters: Record<string, unknown>;
  };
}

export type SearchError =
  | { type: 'QUERY_TOO_SHORT' }
  | { type: 'QUERY_TOO_LONG' }
  | { type: 'INVALID_FILTER'; field: string; value: string };

export type SearchServiceResult =
  | { ok: true; result: SearchResult }
  | { ok: false; error: SearchError };

export async function searchRecords(params: SearchParams): Promise<SearchServiceResult> {
  // Validate query length (F2.1 — min 2 when provided; TechArch: max 500)
  if (params.q !== undefined && params.q !== '') {
    if (params.q.length < 2) return { ok: false, error: { type: 'QUERY_TOO_SHORT' } };
    if (params.q.length > 500) return { ok: false, error: { type: 'QUERY_TOO_LONG' } };
  }

  // Validate maturity filter values
  if (params.maturity) {
    for (const v of params.maturity) {
      if (!VALID_MATURITY_VALUES.includes(v as typeof VALID_MATURITY_VALUES[number])) {
        return { ok: false, error: { type: 'INVALID_FILTER', field: 'maturity', value: v } };
      }
    }
  }

  // Validate reuse_potential
  if (params.reuse_potential && !VALID_REUSE_POTENTIAL.includes(params.reuse_potential as typeof VALID_REUSE_POTENTIAL[number])) {
    return { ok: false, error: { type: 'INVALID_FILTER', field: 'reuse_potential', value: params.reuse_potential } };
  }

  const page = Math.max(1, params.page ?? 1);
  const page_size = Math.min(100, Math.max(1, params.page_size ?? 20));
  const sort = (VALID_SORTS.includes(params.sort as typeof VALID_SORTS[number]) ? params.sort : 'relevance') as RawSearchParams['sort'];

  const rows = await executeSearch({
    query: params.q,
    maturity: params.maturity,
    mission_areas: params.mission_areas,
    technology_areas: params.technology_areas,
    review_statuses: params.review_statuses,
    contributing_offices: params.contributing_offices,
    reuse_potential: params.reuse_potential,
    has_artifacts: params.has_artifacts,
    publication_states: ['published'],  // public always sees published only
    page,
    page_size,
    sort,
  });

  const total = rows.length > 0 ? Number(rows[0].total_count) : 0;
  const records = rows.map(({ rank: _r, total_count: _t, ...rest }) => rest);

  return {
    ok: true,
    result: {
      records,
      meta: {
        page,
        page_size,
        total,
        query: params.q ?? '',
        active_filters: {
          maturity: params.maturity,
          mission_areas: params.mission_areas,
          technology_areas: params.technology_areas,
          review_statuses: params.review_statuses,
          contributing_offices: params.contributing_offices,
          reuse_potential: params.reuse_potential,
          has_artifacts: params.has_artifacts,
        },
      },
    },
  };
}

export async function getFacetCounts(): Promise<FacetCounts> {
  return executeFacetCounts();
}
```

```typescript
import { NextResponse } from 'next/server';
import { searchRecords } from '@/lib/services/search.service';

function parseArray(param: string | string[] | null): string[] | undefined {
  if (!param) return undefined;
  if (Array.isArray(param)) return param;
  return [param];
}

export async function GET(request: Request) {
  const { searchParams } = new URL(request.url);

  const params = {
    q: searchParams.get('q') ?? undefined,
    mission_areas: parseArray(searchParams.getAll('mission_areas[]')),
    technology_areas: parseArray(searchParams.getAll('technology_areas[]')),
    maturity: parseArray(searchParams.getAll('maturity[]')),
    review_statuses: parseArray(searchParams.getAll('review_statuses[]')),
    contributing_offices: parseArray(searchParams.getAll('contributing_offices[]')),
    reuse_potential: searchParams.get('reuse_potential') ?? undefined,
    has_artifacts: searchParams.get('has_artifacts') === 'true' ? true : searchParams.get('has_artifacts') === 'false' ? false : undefined,
    page: searchParams.get('page') ? Number(searchParams.get('page')) : undefined,
    page_size: searchParams.get('page_size') ? Number(searchParams.get('page_size')) : undefined,
    sort: searchParams.get('sort') ?? undefined,
  };

  const outcome = await searchRecords(params);

  if (!outcome.ok) {
    const statusMap: Record<string, number> = {
      QUERY_TOO_SHORT: 400,
      QUERY_TOO_LONG: 400,
      INVALID_FILTER: 400,
    };
    return NextResponse.json(
      { status: 'error', error_code: outcome.error.type, message: outcome.error.type },
      { status: statusMap[outcome.error.type] ?? 400 }
    );
  }

  return NextResponse.json({
    status: 'ok',
    data: outcome.result.records,
    meta: outcome.result.meta,
  });
}
```
