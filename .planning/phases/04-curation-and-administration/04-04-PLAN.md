**src/app/api/v1/curator/records/[id]/audit/route.ts** — F9.11 chronological audit history:

```typescript
import { NextResponse } from 'next/server';
import { requireRole } from '@/lib/auth/middleware';
import { db } from '@/lib/db/client';

export async function GET(request: Request, { params }: { params: { id: string } }) {
  const auth = await requireRole(request, 'curator');
  if (auth instanceof Response) return auth;

  const { searchParams } = new URL(request.url);
  const page = Math.max(1, Number(searchParams.get('page') ?? 1));
  const pageSize = Math.min(100, Number(searchParams.get('page_size') ?? 50));

  // Audit history is chronological, append-only (never modified)
  const [events, totalRow] = await Promise.all([
    db.selectFrom('audit_events')
      .select(['audit_id', 'event_type', 'actor_name', 'event_data', 'target_title', 'occurred_at', 'notes'])
      .where('target_id', '=', params.id)
      .orderBy('occurred_at', 'asc')  // chronological — oldest first
      .limit(pageSize)
      .offset((page - 1) * pageSize)
      .execute(),
    db.selectFrom('audit_events')
      .select(db.fn.count('audit_id').as('count'))
      .where('target_id', '=', params.id)
      .executeTakeFirst(),
  ]);

  return NextResponse.json({
    status: 'ok',
    data: events,
    meta: { page, page_size: pageSize, total: Number(totalRow?.count ?? 0) },
  });
}
```

```typescript
import { NextResponse } from 'next/server';
import { requireRole } from '@/lib/auth/middleware';
import { db } from '@/lib/db/client';
import { appendAuditEvent } from '@/lib/services/audit.service';
import { z } from 'zod';

const VALID_STATUSES = ['accepted', 'declined', 'needs_more_information', 'duplicate'] as const;

const DispositionSchema = z.object({
  status: z.enum(VALID_STATUSES),
  curator_notes: z.string().max(3000).optional(),
});

export async function PATCH(request: Request, { params }: { params: { id: string } }) {
  const auth = await requireRole(request, 'curator');
  if (auth instanceof Response) return auth;

  const body = await request.json().catch(() => null);
  const parsed = DispositionSchema.safeParse(body);
  if (!parsed.success) {
    return NextResponse.json({ status: 'error', error_code: 'VALIDATION_ERROR', message: 'Invalid disposition data' }, { status: 422 });
  }

  await db
    .updateTable('opportunity_submissions')
    .set({
      status: parsed.data.status,
      curator_notes: parsed.data.curator_notes ?? null,
      dispositioned_at: new Date().toISOString() as unknown as Date,
      dispositioned_by: auth.session.userId as unknown as null,  // cast for type compat
    })
    .where('id', '=', params.id)
    .execute();

  await appendAuditEvent({
    eventType: 'submission_dispositioned',
    actorId: auth.session.userId,
    actorName: auth.session.name,
    targetType: 'opportunity_submission',
    targetId: params.id,
    eventData: { status: parsed.data.status, hasNotes: !!parsed.data.curator_notes },
  });

  return NextResponse.json({ status: 'ok' });
}
```

**src/app/api/v1/curator/submissions/contribution/[id]/create-record/route.ts** — F9.13 creates draft from contribution (per TechArch §4.4):

```typescript
import { NextResponse } from 'next/server';
import { requireRole } from '@/lib/auth/middleware';
import { db } from '@/lib/db/client';
import { appendAuditEvent } from '@/lib/services/audit.service';

export async function POST(request: Request, { params }: { params: { id: string } }) {
  const auth = await requireRole(request, 'curator');
  if (auth instanceof Response) return auth;

  // Fetch the contribution
  const contribution = await db
    .selectFrom('innovation_contributions')
    .selectAll()
    .where('id', '=', params.id)
    .executeTakeFirst();

  if (!contribution) {
    return NextResponse.json({ status: 'error', error_code: 'NOT_FOUND', message: 'Contribution not found' }, { status: 404 });
  }
  if (contribution.created_record_id) {
    return NextResponse.json({ status: 'error', error_code: 'ALREADY_CURATED', message: 'A record was already created from this contribution.' }, { status: 409 });
  }

  const recordId = crypto.randomUUID();
  const slug = contribution.contribution_title.toLowerCase().replace(/[^a-z0-9\s-]/g, '').replace(/\s+/g, '-').slice(0, 100) + '-' + Date.now().toString(36);

  // Create draft record pre-populated from contribution (per TechArch §4.4)
  await db.insertInto('innovation_records').values({
    id: recordId,
    slug,
    publication_state: 'draft',
    created_by: auth.session.userId,
    updated_by: auth.session.userId,
    title: contribution.contribution_title,
    summary: '',
    problem_statement: contribution.problem_addressed,
    hypothesis_or_objective: '',
    outcome_summary: '',
    source_basis: '',
    // F7.3 attribution fields preserved from contribution
    contributing_offices: [contribution.contributing_office],
    contributor_names: [contribution.contributor_names],
    owner_steward: contribution.current_owner,
    owner_contact: contribution.owner_contact_email,
    attribution_statement: `Contributed by ${contribution.contributor_names} from ${contribution.contributing_office}.`,
    applicable_disclaimer: '',
    source_contribution_id: params.id,  // immutable link back to contribution (F7.3)
    maturity: contribution.current_maturity as 'idea',
    engagement_indicator: 'none',
  }).execute();

  // Update contribution to mark as curated
  await db
    .updateTable('innovation_contributions')
    .set({ status: 'curated', created_record_id: recordId, dispositioned_at: new Date().toISOString() as unknown as Date, dispositioned_by: auth.session.userId as unknown as null })
    .where('id', '=', params.id)
    .execute();

  await appendAuditEvent({
    eventType: 'record_created_from_contribution',
    actorId: auth.session.userId,
    actorName: auth.session.name,
    targetType: 'innovation_record',
    targetId: recordId,
    targetTitle: contribution.contribution_title,
    eventData: { contribution_id: params.id, contributing_office: contribution.contributing_office },
  });

  return NextResponse.json({
    status: 'ok',
    data: { recordId, message: 'Draft record created from contribution. Attribution fields pre-populated.' },
  }, { status: 201 });
}
```

```typescript
import { NextResponse } from 'next/server';
import { requireRole } from '@/lib/auth/middleware';
import { db } from '@/lib/db/client';
import { appendAuditEvent } from '@/lib/services/audit.service';
import { z } from 'zod';

const FOLLOW_UP_STATUSES = ['received', 'in_progress', 'completed', 'no_action_required'] as const;
const StatusSchema = z.object({
  follow_up_status: z.enum(FOLLOW_UP_STATUSES),
  curator_notes: z.string().max(3000).optional(),
});

export async function PATCH(request: Request, { params }: { params: { id: string } }) {
  const auth = await requireRole(request, 'curator');
  if (auth instanceof Response) return auth;

  const body = await request.json().catch(() => null);
  const parsed = StatusSchema.safeParse(body);
  if (!parsed.success) {
    return NextResponse.json({ status: 'error', error_code: 'VALIDATION_ERROR' }, { status: 422 });
  }

  await db
    .updateTable('engagement_requests')
    .set({
      follow_up_status: parsed.data.follow_up_status,
      curator_notes: parsed.data.curator_notes ?? null,
      follow_up_updated_at: new Date().toISOString() as unknown as Date,
      follow_up_updated_by: auth.session.userId as unknown as null,
    })
    .where('id', '=', params.id)
    .execute();

  await appendAuditEvent({
    eventType: 'engagement_status_updated',
    actorId: auth.session.userId,
    actorName: auth.session.name,
    targetType: 'engagement_request',
    targetId: params.id,
    eventData: { status: parsed.data.follow_up_status },
  });

  return NextResponse.json({ status: 'ok' });
}
```

```typescript
import { NextResponse } from 'next/server';
import { requireRole } from '@/lib/auth/middleware';
import { db } from '@/lib/db/client';
import { appendAuditEvent } from '@/lib/services/audit.service';

// Admin-only (AUTH-03)
export async function PUT(request: Request, { params }: { params: { key: string } }) {
  const auth = await requireRole(request, 'admin');  // Admin only — F9.15
  if (auth instanceof Response) return auth;

  const body = await request.json().catch(() => null);
  if (!body || typeof body.value !== 'string') {
    return NextResponse.json({ status: 'error', error_code: 'VALIDATION_ERROR', message: 'value field required' }, { status: 422 });
  }

  const existing = await db.selectFrom('hub_settings').select(['setting_key', 'setting_value']).where('setting_key', '=', params.key).executeTakeFirst();
  if (!existing) {
    return NextResponse.json({ status: 'error', error_code: 'NOT_FOUND', message: `Setting '${params.key}' not found.` }, { status: 404 });
  }

  await db.updateTable('hub_settings').set({ setting_value: body.value, updated_at: new Date().toISOString() as unknown as Date, updated_by: auth.session.userId as unknown as null }).where('setting_key', '=', params.key).execute();

  await appendAuditEvent({
    eventType: 'settings_changed',
    actorId: auth.session.userId,
    actorName: auth.session.name,
    targetType: 'hub_settings',
    targetId: params.key,
    eventData: { key: params.key, previousValue: existing.setting_value, newValue: body.value },
  });

  // Invalidate cache in hub-settings.service.ts — clear the cached entry
  // (cache.delete called in hub-settings.service; here just return success)
  return NextResponse.json({ status: 'ok' });
}
```
